#!/bin/bash

# =============================================================================
#  Скрипт первичной настройки сервера Debian 12
# =============================================================================

set -e  # Остановка при любой ошибке

# --- Цвета для вывода ---
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

log() { echo -e "${GREEN}[✓]${NC} $1"; }
warn() { echo -e "${YELLOW}[!]${NC} $1"; }
err() { echo -e "${RED}[✗]${NC} $1"; }

# =============================================================================
# 1. ОБНОВЛЕНИЕ СИСТЕМЫ
# =============================================================================
log "Обновление списков пакетов и системы..."
apt update -y && apt upgrade -y
log "Система обновлена."

# =============================================================================
# 2. БАЗОВЫЕ ПАКЕТЫ
# =============================================================================
log "Установка базовых пакетов..."
apt install -y \
    curl \
    wget \
    nano \
    ufw \
    fail2ban \
    unzip \
    socat \
    cron \
    ca-certificates \
    openssl \
    tzdata \
    net-tools \
    htop \
    jq
log "Базовые пакеты установлены."

# =============================================================================
# 3. ЧАСОВОЙ ПОЯС
# =============================================================================
log "Установка часового пояса Europe/Moscow..."
timedatectl set-timezone Europe/Moscow
log "Часовой пояс: $(timedatectl show --property=Timezone --value)"

# =============================================================================
# 4. УСКОРЕНИЕ ИНТЕРНЕТА (BBR + оптимизация TCP)
# =============================================================================
log "Включение BBR и оптимизация сетевого стека..."

cat >> /etc/sysctl.conf << 'EOF'

# --- Network Optimization (BBR + TCP tuning) ---
net.core.default_qdisc = fq
net.ipv4.tcp_congestion_control = bbr
net.core.rmem_max = 26214400
net.core.wmem_max = 26214400
net.ipv4.tcp_rmem = 4096 87380 26214400
net.ipv4.tcp_wmem = 4096 65536 26214400
net.ipv4.tcp_fastopen = 3
net.ipv4.tcp_slow_start_after_idle = 0
net.ipv4.tcp_mtu_probing = 1
net.ipv4.tcp_notsent_lowat = 16384
vm.swappiness = 10
EOF

sysctl -p > /dev/null 2>&1
log "BBR активирован: $(sysctl -n net.ipv4.tcp_congestion_control)"

# =============================================================================
# 5. ФАЙРВОЛ (UFW)
# =============================================================================
log "Настройка файрвола UFW..."
ufw default deny incoming
ufw default allow outgoing
ufw allow 22/tcp comment 'SSH'
ufw --force enable
log "UFW включён. Статус:"
ufw status verbose

# =============================================================================
# 6. FAIL2BAN (защита от перебора)
# =============================================================================
log "Настройка Fail2Ban..."

cat > /etc/fail2ban/jail.local << 'EOF'
[DEFAULT]
bantime  = 3600
findtime = 600
maxretry = 5
backend  = systemd

[sshd]
enabled = true
port    = 22
filter  = sshd
logpath = %(sshd_log)s
EOF

systemctl enable fail2ban
systemctl restart fail2ban
log "Fail2Ban запущен. Статус:"
systemctl status fail2ban --no-pager | head -5

# =============================================================================
# 7. ОТКЛЮЧЕНИЕ УВЕДОМЛЕНИЙ ПРИ ВХОДЕ (MOTD)
# =============================================================================
log "Отключение приветственных сообщений при входе по SSH..."

# Для текущего root-пользователя
touch ~/.hushlogin

# Отключаем motd-news таймер (Debian/Ubuntu)
systemctl disable --now motd-news.timer 2>/dev/null || true

# Отключаем динамический MOTD
chmod -x /etc/update-motd.d/* 2>/dev/null || true

# Очищаем статический MOTD
[ -f /etc/motd ] && cp /etc/motd /etc/motd.bak && echo > /etc/motd

log "Уведомления при входе отключены."

# =============================================================================
# 8. ОЧИСТКА СИСТЕМЫ
# =============================================================================
log "Очистка системы..."
apt clean
apt autoremove --purge -y
log "Система очищена от кэша и ненужных пакетов."

# =============================================================================
# 9. ОТКЛЮЧЕНИЕ ЛОГИРОВАНИЯ
# =============================================================================
warn "ОТКЛЮЧЕНИЕ ЛОГИРОВАНИЯ: Это действие может усложнить диагностику проблем."
warn "Если вы не уверены — закомментируйте этот блок и перезапустите скрипт."

# --- 9a. Отключаем rsyslog (традиционный syslog) ---
if systemctl list-unit-files | grep -q rsyslog.service; then
    systemctl stop rsyslog
    systemctl disable rsyslog
    log "rsyslog остановлен и отключён."
fi

# --- 9b. Отключаем persistent-логирование systemd-journald ---
mkdir -p /etc/systemd/journald.conf.d
cat > /etc/systemd/journald.conf.d/discard.conf << 'EOF'
[Journal]
Storage=none
ForwardToSyslog=no
ForwardToKMsg=no
ForwardToWall=no
EOF

systemctl restart systemd-journald
log "journald переведён в режим 'Storage=none' (без записи на диск)."

# --- 9c. Очищаем уже накопленные логи ---
journalctl --vacuum-time=1s 2>/dev/null || true
rm -f /var/log/syslog /var/log/messages /var/log/auth.log /var/log/kern.log 2>/dev/null || true
log "Накопленные логи очищены."

# =============================================================================
# 10. УСТАНОВКА 3X-UI
# =============================================================================
log "Установка панели 3x-ui..."

# Официальный установочный скрипт (MHSanaei)
bash <(curl -Ls https://raw.githubusercontent.com/mhsanaei/3x-ui/master/install.sh) <<< "y"

log "3x-ui установлен."

# =============================================================================
# ИТОГОВАЯ ИНФОРМАЦИЯ
# =============================================================================
echo ""
echo "======================================================================"
echo -e "${GREEN}  НАСТРОЙКА СЕРВЕРА ЗАВЕРШЕНА${NC}"
echo "======================================================================"
echo ""
echo "  Что сделано:"
echo "   ✓ Система обновлена"
echo "   ✓ Базовые пакеты установлены"
echo "   ✓ Часовой пояс: Europe/Moscow"
echo "   ✓ BBR + TCP-оптимизация"
echo "   ✓ UFW: разрешён только SSH (порт 22)"
echo "   ✓ Fail2Ban: защита SSH"
echo "   ✓ MOTD и приветствия отключены"
echo "   ✓ Кэш APT и мусор удалены"
echo "   ✓ rsyslog и journald отключены"
echo "   ✓ 3x-ui установлен"
echo ""
echo "  ВАЖНО:"
echo "   • Управление 3x-ui: команда 'x-ui'"
echo "   • Откройте порт панели в UFW: ufw allow <порт>/tcp"
echo "   • Создайте SSH-ключ и отключите вход по паролю!"
echo "   • Перезагрузите сервер: reboot"
echo "======================================================================"
